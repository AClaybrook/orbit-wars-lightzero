  --append-cache --games-2p 16 --games-4p 8 --workers 2 --collect-only
python scripts/train_planet_transformer.py --dataset-cache pt_dataset_gen6_bc.npz --save-cache pt_dataset_gen6.npz \
  --dagger-checkpoint checkpoints/planet_transformer_v3_rl_latest.pt --dagger-2p 8 --dagger-4p 4 --workers 2 --collect-only

# 2. Train 1.89M PlanetTimelineTransformer (d=192, h=6, L=6, ff=384)
python scripts/train_planet_transformer.py --dataset-cache pt_dataset_gen6.npz \
  --d-model 192 --n-heads 6 --n-layers 6 --dim-ff 384 --epochs 15 --batch-size 96 --lr 1.5e-3 \
  --out checkpoints/planet_transformer_v4_large.pt
